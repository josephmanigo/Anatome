cmake_minimum_required(VERSION 3.18)
project(ThoraxCut LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MANIFOLD_TEST OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CBIND OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CROSS_SECTION OFF CACHE BOOL "" FORCE)
set(MANIFOLD_PAR OFF CACHE BOOL "" FORCE)
set(MANIFOLD_PYBIND OFF CACHE BOOL "" FORCE)
set(MANIFOLD_DOWNLOADS OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/manifold)
add_library(thoraxcut SHARED thoraxcut.cpp)
target_link_libraries(thoraxcut PRIVATE manifold)
target_include_directories(thoraxcut PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(ANDROID)
  target_link_options(thoraxcut PRIVATE --driver-mode=g++ "-Wl,-z,max-page-size=16384")
else()
  add_executable(thoraxcut_tests tests.cpp)
  target_link_libraries(thoraxcut_tests PRIVATE thoraxcut manifold)
  enable_testing()
  add_test(NAME thoraxcut-contract COMMAND thoraxcut_tests)
endif()
